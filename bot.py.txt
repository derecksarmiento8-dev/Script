import discord
from discord.ext import commands
import json
import os

intents = discord.Intents.default()
bot = commands.Bot(command_prefix="!", intents=intents)

DATA_FILE = "scripts.json"

# 📁 Load / Save
def load_scripts():
    if not os.path.exists(DATA_FILE):
        return {}
    with open(DATA_FILE, "r") as f:
        return json.load(f)

def save_scripts(data):
    with open(DATA_FILE, "w") as f:
        json.dump(data, f, indent=4)

scripts = load_scripts()

# 🔽 Dropdown menu
class ScriptSelect(discord.ui.Select):
    def __init__(self):
        options = []

        for name in scripts.keys():
            options.append(discord.SelectOption(label=name))

        if not options:
            options.append(discord.SelectOption(label="No scripts yet"))

        super().__init__(
            placeholder="Select a script...",
            options=options
        )

    async def callback(self, interaction: discord.Interaction):
        name = self.values[0]

        if name not in scripts:
            await interaction.response.send_message("No script found.", ephemeral=True)
            return

        await interaction.response.send_message(
            f"```{scripts[name]}```",
            ephemeral=True
        )

class ScriptView(discord.ui.View):
    def __init__(self):
        super().__init__(timeout=None)  # 🔥 stays forever
        self.add_item(ScriptSelect())

# 📌 PERMANENT PANEL COMMAND
@bot.tree.command(name="panel", description="Create permanent script panel")
async def panel(interaction: discord.Interaction):
    await interaction.channel.send(
        "📜 **Script Panel**\nSelect a script below:",
        view=ScriptView()
    )

    await interaction.response.send_message(
        "Panel created in this channel.",
        ephemeral=True
    )

# ➕ Add script
@bot.tree.command(name="addscript", description="Add a script")
async def addscript(interaction: discord.Interaction, name: str, content: str):
    scripts[name] = content
    save_scripts(scripts)
    await interaction.response.send_message(f"Saved **{name}**", ephemeral=True)

# 🗑️ Delete script
@bot.tree.command(name="deletescript", description="Delete a script")
async def deletescript(interaction: discord.Interaction, name: str):
    if name in scripts:
        del scripts[name]
        save_scripts(scripts)
        await interaction.response.send_message(f"Deleted **{name}**", ephemeral=True)
    else:
        await interaction.response.send_message("Not found.", ephemeral=True)

# 🚀 Bot start
@bot.event
async def on_ready():
    await bot.tree.sync()
    print(f"Logged in as {bot.user}")
